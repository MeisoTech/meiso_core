cmake_minimum_required(VERSION 3.21)
project(MeisoLegacyV12Rejection LANGUAGES C)
find_package(MeisoGlass 0.1 REQUIRED CONFIG COMPONENTS Profiles)
if(NOT MeisoGlass_SOURCE_REVISION STREQUAL "d2882982709b372a975ca91d6c2c62bb5b24bb1f")
    message(FATAL_ERROR "Use the unmodified pre-extension SDK d288298, got ${MeisoGlass_SOURCE_REVISION}")
endif()
add_executable(meiso_legacy_v12_rejection main.c)
set_target_properties(meiso_legacy_v12_rejection PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_link_libraries(meiso_legacy_v12_rejection PRIVATE Meiso::DisplayIngressProfile)
if(MSVC)
    target_compile_options(meiso_legacy_v12_rejection PRIVATE /W4 /WX)
else()
    target_compile_options(meiso_legacy_v12_rejection PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME meiso_legacy_v12_rejection COMMAND meiso_legacy_v12_rejection)
